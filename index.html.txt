        function quickLogin(role) {
            selectLoginRole(role);
            const userId = document.getElementById('loginIdInput').value;
            const user = appState.users.find(u => u.id === userId && u.role === role);
            if(user) loginUser(user);
        }

        function handleLoginSubmit(e) {
            e.preventDefault();
            const id = document.getElementById('loginIdInput').value.trim();
            const pass = document.getElementById('loginPasswordInput').value.trim();
            const role = appState.activeLoginRole;

            const user = appState.users.find(u => u.id === id && u.role === role);
            if(!user) {
                showToast(`Invalid ${role.toUpperCase()} ID or account non-existent`, 'error');
                return;
            }

            if(user.password !== pass) {
                showToast(`Incorrect Password! Use 'Forgot password?' to reset.`, 'error');
                return;
            }

            loginUser(user);
        }

        function loginUser(userObj) {
            appState.currentUser = userObj;
            appState.activeTab = 'dashboard';

            document.getElementById('loginScreen').classList.add('hidden');
            document.getElementById('appContainer').classList.remove('hidden');

            document.getElementById('headerUserName').innerText = userObj.name;
            document.getElementById('headerUserRole').innerText = userObj.role;
            document.getElementById('headerUserAvatar').src = userObj.avatar;

            const btnEnroll = document.getElementById('btnHeaderEnroll');
            if(userObj.role === 'admin') {
                btnEnroll.classList.remove('hidden');
            } else {
                btnEnroll.classList.add('hidden');
            }

            renderSidebar();
            renderMainContent();
            showToast(`Welcome back, ${userObj.name}!`, 'success');
        }

        function startAdminEnrollCamera() {
            const video = document.getElementById('enrollWebcam');
            const placeholder = document.getElementById('enrollCameraPlaceholder');
            if(navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
                navigator.mediaDevices.getUserMedia({ video: true })
                    .then(stream => {
                        appState.adminCameraStream = stream;
                        video.srcObject = stream;
                        video.classList.remove('hidden');
                        placeholder.classList.add('hidden');
                        showToast('Live camera feed active', 'info');
                    })
                    .catch(err => {
                        showToast('Unable to access webcam. Choose photo file instead.', 'error');
                    });
            }
        }

        function stopAdminEnrollCamera() {
            if(appState.adminCameraStream) {
                appState.adminCameraStream.getTracks().forEach(t => t.stop());
                appState.adminCameraStream = null;
            }
            const video = document.getElementById('enrollWebcam');
            const placeholder = document.getElementById('enrollCameraPlaceholder');
            if(video) video.classList.add('hidden');
            if(placeholder) placeholder.classList.remove('hidden');
        }

        function captureAdminEnrollPhoto() {
            const video = document.getElementById('enrollWebcam');
            if(!appState.adminCameraStream || video.classList.contains('hidden')) {
                showToast('Start camera stream first!', 'error');
                return;
            }
            const canvas = document.getElementById('hiddenSnapshotCanvas');
            canvas.width = video.videoWidth || 300;
            canvas.height = video.videoHeight || 300;
            const ctx = canvas.getContext('2d');
            ctx.drawImage(video, 0, 0, canvas.width, canvas.height);
            
            appState.capturedAdminPhotoUrl = canvas.toDataURL('image/jpeg');
            document.getElementById('adminPhotoPreview').src = appState.capturedAdminPhotoUrl;
            showToast('Student face photo registered!', 'success');
        }

        function previewAdminPhotoFile(e) {
            const file = e.target.files[0];
            if(file) {
                const reader = new FileReader();
                reader.onload = function(evt) {
                    appState.capturedAdminPhotoUrl = evt.target.result;
                    document.getElementById('adminPhotoPreview').src = evt.target.result;
                }
                reader.readAsDataURL(file);
            }
        }

            showToast('All students marked Present', 'success');
            renderFacultyLiveSession(document.getElementById('mainContent'));
        }

        function startFacultyLiveEngine() {
            updateDynamicQr();
            appState.liveSession.timer = 30;
            if(appState.liveSession.intervalId) clearInterval(appState.liveSession.intervalId);
            
            appState.liveSession.intervalId = setInterval(() => {
                appState.liveSession.timer--;
                const timerEl = document.getElementById('qrTimer');
                if(timerEl) timerEl.innerText = `${appState.liveSession.timer}s`;

                if(appState.liveSession.timer <= 0) {
                    appState.liveSession.timer = 30;
                    updateDynamicQr();
                }
            }, 1000);

            const video = document.getElementById('facultyWebcam');
            if(video && navigator.mediaDevices && navigator.mediaDevices.getUserMedia) {
                navigator.mediaDevices.getUserMedia({ video: true })
                    .then(stream => {
                        appState.liveSession.webcamStream = stream;
                        video.srcObject = stream;
                    })
                    .catch(err => {
                        showToast('Camera simulation mode active', 'info');
                    });
            }
        }

        function updateDynamicQr() {
            const qrContainer = document.getElementById('qrCodeContainer');
            if(!qrContainer) return;
            qrContainer.innerHTML = '';

            const student = appState.users.find(u => u.id === appState.liveSession.selectedStudentId) || appState.users[2];
            appState.liveSession.token = `MU_PYTHON_STU_${student.id}_${Date.now()}`;

            new QRCode(qrContainer, {
                text: appState.liveSession.token,
                width: 140,
                height: 140,
                colorDark : "#0f172a",
                colorLight : "#ffffff",
                correctLevel : QRCode.CorrectLevel.H
            });
        }

        function stopLiveSession() {
            if(appState.liveSession.intervalId) {
                clearInterval(appState.liveSession.intervalId);
                appState.liveSession.intervalId = null;
            }
            if(appState.liveSession.webcamStream) {
                appState.liveSession.webcamStream.getTracks().forEach(track => track.stop());
                appState.liveSession.webcamStream = null;
            }
        }

        function triggerManualFaceMatch() {
            const students = appState.users.filter(u => u.role === 'student');
            const randomStudent = students[Math.floor(Math.random() * students.length)];
            if(randomStudent) {
                randomStudent.status = 'Present';
                appState.liveSession.lastMarkedStudent = randomStudent;

                const box = document.getElementById('faceOverlayBox');
                const nameLabel = document.getElementById('matchedStudentName');
                const badge = document.getElementById('faceStatusBadge');

                if(box) box.classList.add('matched');
                if(nameLabel) nameLabel.innerText = `Matched: ${randomStudent.name}`;
                if(badge) {
                    badge.innerText = `Verified: ${randomStudent.name}`;
                    badge.className = 'text-[10px] font-bold text-emerald-400 bg-emerald-400/10 px-2 py-0.5 rounded-md border border-emerald-400/20';
                }

                showToast(`Face Matched: ${randomStudent.name} marked Present!`, 'success');
                renderFacultyLiveSession(document.getElementById('mainContent'));
            }
        }

        function toggleCurriculumTopic(id) {
            const item = appState.curriculum.find(c => c.id === id);
            if(item) {
                item.completed = !item.completed;
                item.date = item.completed ? new Date().toISOString().split('T')[0] : 'Pending';
                renderCurriculumModule(document.getElementById('mainContent'));
                showToast(`Curriculum topic status updated`, 'info');
            }
        }

        function handleActivitySubmit(e) {
            e.preventDefault();
            const title = document.getElementById('actTitle').value;
            const category = document.getElementById('actCategory').value;

            appState.activities.unshift({
                id: Date.now(),
                studentId: appState.currentUser.id,
                studentName: appState.currentUser.name,
                title: title,
                category: category,
                date: new Date().toISOString().split('T')[0],
                status: 'Approved'
            });

            showToast('Activity record logged', 'success');
            renderActivitiesModule(document.getElementById('mainContent'));
        }

        // --- TOAST NOTIFICATIONS ---
        function showToast(msg, type = 'info') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');

            const colors = {
                success: 'bg-emerald-950 border-emerald-700 text-emerald-200',
                error: 'bg-rose-950 border-rose-700 text-rose-200',
                info: 'bg-slate-900 border-slate-700 text-slate-200'
            };

            toast.className = `px-4 py-2.5 rounded-xl border text-xs font-medium shadow-xl transition-all duration-300 pointer-events-auto transform translate-y-2 opacity-0 ${colors[type]}`;
            toast.innerText = msg;

            container.appendChild(toast);

            setTimeout(() => {
                toast.classList.remove('translate-y-2', 'opacity-0');
            }, 10);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
